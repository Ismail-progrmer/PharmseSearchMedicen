CREATE TABLE Categories (
CategoryID NUMBER ,
CategoryName VARCHAR2(200) NOT NULL,
 CONSTRAINT PK_Categories
       PRIMARY KEY (CategoryID)
);


CREATE TABLE Medicines (
    MedicineID NUMBER,
MedicineName VARCHAR2(200) NOT NULL,
    Strength VARCHAR2(200),
    DosageForm VARCHAR2(50),
    CategoryID NUMBER(10),
 CONSTRAINT PK_Medicines
        PRIMARY KEY (MedicineID),
   CONSTRAINT FK_Medicines_Categories
        FOREIGN KEY (CategoryID)
        REFERENCES Categories(CategoryID)
);
CREATE TABLE Pharmacies (
    PharmaciesID NUMBER,
   PharmacyName VARCHAR2(100) NOT NULL,
    UserName VARCHAR2(200) NOT NULL,
    Password VARCHAR2(200) NOT NULL,
  Address VARCHAR2(200),
    Phone VARCHAR2(50),
   CONSTRAINT PK_Pharmacies
        PRIMARY KEY (PharmaciesID),

    CONSTRAINT UQ_Pharmacies_UserName
        UNIQUE (UserName)
);


CREATE TABLE WorkingHours (
    WorkingHoursID NUMBER,
    DayOfWeek VARCHAR2(20) NOT NULL,
    OpenTime DATE,
    CloseTime DATE,
    IsOpen NUMBER(1) DEFAULT 1 NOT NULL,
    PharmaciesID NUMBER NOT NULL,
    CONSTRAINT FK_WorkingHours_Pharmacies
        FOREIGN KEY (PharmaciesID)
        REFERENCES Pharmacies(PharmaciesID),

    CONSTRAINT UQ_Pharmacy_Day
        UNIQUE (PharmaciesID, DayOfWeek),

    CONSTRAINT CK_WorkingHours_IsOpen
        CHECK (IsOpen IN (0, 1))
);

CREATE TABLE PharmacyMedicines (
    PharmaciesID NUMBER(10) NOT NULL,
    MedicineID NUMBER(10) NOT NULL,
    Price DECIMAL(19,2) NOT NULL,
    Quantity NUMBER(10) NOT NULL,
    LastUpdate DATE DEFAULT SYSDATE NOT NULL,

    CONSTRAINT PK_PharmacyMedicines
        PRIMARY KEY (PharmaciesID, MedicineID),

    CONSTRAINT FK_PM_Pharmacies
        FOREIGN KEY (PharmaciesID)
        REFERENCES Pharmacies(PharmaciesID),

    CONSTRAINT FK_PM_Medicines
        FOREIGN KEY (MedicineID)
        REFERENCES Medicines(MedicineID),

    CONSTRAINT CK_PM_Price
        CHECK (Price >= 0),

    CONSTRAINT CK_PM_Quantity
        CHECK (Quantity >= 0)
);
